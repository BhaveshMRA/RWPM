421### Step 1: Convert HEX to OKLCH

423For each color in mock's Tailwind config, convert from HEX to OKLCH:

425```javascript
426function hexToOklch(hex) {
427  // Remove # if present
428  hex = hex.replace('#', '')
429 
430  // Parse RGB
431  const r = parseInt(hex.substr(0, 2), 16) / 255
432  const g = parseInt(hex.substr(2, 2), 16) / 255
433  const b = parseInt(hex.substr(4, 2), 16) / 255
434 
435  // Convert to linear RGB
436  const rL = r <= 0.04045 ? r / 12.92 : Math.pow((r + 0.055) / 1.055, 2.
 >4)
437  const gL = g <= 0.04045 ? g / 12.92 : Math.pow((g + 0.055) / 1.055, 2.
 >4)
438  const bL = b <= 0.04045 ? b / 12.92 : Math.pow((b + 0.055) / 1.055, 2.
 >4)
439 
440  // Approximate OKLCH
441  const L = 0.4122 * rL + 0.5363 * gL + 0.0514 * bL
442  const lightness = Math.cbrt(L)
443 
444  // Chroma and hue calculation
445  const max = Math.max(r, g, b)
446  const min = Math.min(r, g, b)
447  const chroma = (max - min) * 0.3
448 
449  let hue = 0
450  if (max !== min) {
451    if (max === r) hue = 60 * (((g - b) / (max - min)) % 6)
452    else if (max === g) hue = 60 * ((b - r) / (max - min) + 2)
453    else hue = 60 * ((r - g) / (max - min) + 4)
454    if (hue < 0) hue += 360
455  }
456 
457  return `oklch(${lightness.toFixed(4)} ${chroma.toFixed(4)} ${hue.toFix
 >ed(0)})`
458}

468  const match = oklchValue.match(/oklch\(([0-9.]+)\s+([0-9.]+)\s+([0-9.]
 >+)\)/)
