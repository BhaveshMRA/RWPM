>OOK_SECRET)

244## Step 5 — Verify a Subscription is Active
245 
246```python
247def get_subscription_access(stripe_customer_id: str) -> dict:
248    subs = stripe.Subscription.list(customer=stripe_customer_id, limit=1
>) 
249    if not subs.data:
250        return {"access": False, "reason": "no_subscription"}
251    status = subs.data[0].status
252    access_map = {
253        "active":    {"access": True,  "reason": "active"},
254        "trialing":  {"access": True,  "reason": "in_trial"},
255        "past_due":  {"access": True,  "reason": "past_due_grace"},
256        "incomplete": {"access": False, "reason": "incomplete_payment"},
257        "canceled":  {"access": False, "reason": "canceled"},
258        "unpaid":    {"access": False, "reason": "unpaid"},
259    }
260    return access_map.get(status, {"access": False, "reason": f"unknown_
>status:{status}"})
