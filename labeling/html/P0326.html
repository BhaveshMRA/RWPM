89## 三、Store 标准结构
90 
91### 标准写法（Setup Store 风格，与 `<script setup>` 一致）
92 
93```typescript
94// stores/user.ts
95import type { UserInfo, Permission } from '@/types/user'
96 
97export const useUserStore = defineStore('user', () => {
98  // ① State —— 响应式数据
99  const userInfo = ref<UserInfo | null>(null)
100  const permissions = ref<string[]>([])
101  const token = useStorage('token', '')
102 
103  // ② Getters —— 派生数据
104  const isLoggedIn = computed(() => !!token.value)
105  const hasPermission = computed(
106    () => (code: string) => permissions.value.includes(code)
107  )
108 
109  // ③ Actions —— 方法（含异步）
110  async function fetchUserInfo() {
111    const { data } = await axios.get('/api/user/info')
112    if (data?.code !== 200) return
113    userInfo.value = data.data
114    permissions.value = data.data.permissions ?? []
115  }
116 
117  function logout() {
118    userInfo.value = null
119    permissions.value = []
120    token.value = ''
121  }
122 
123  return {
124    // 暴露 state（用 readonly 保护不应被外部直接修改的）
125    userInfo: readonly(userInfo),
126    permissions: readonly(permissions),
127    token,
128    // 暴露 getters
129    isLoggedIn,
130    hasPermission,
131    // 暴露 actions
132    fetchUserInfo,
133    logout,
134  }
135})
136```

155## 四、异步 Action 规范
156 
157Store 中的异步 action 与组件中的请求写法保持一致，遵循 `api-integration`：
158 
159```typescript
160// ✅ 标准异步 action
161async function fetchDictList() {
162  dictLoading.value = true
163  try {
164    const { data } = await axios.get('/api/dict/list')
165    if (data?.code !== 200) return
166    dictMap.value = Object.fromEntries(
167      data.data.map((item: DictItem) => [item.code, item])
168    )
169  } finally {
170    dictLoading.value = false
171  }
172}
173 
174// ❌ 禁止 —— 没有 finally，loading 漏关
175async function badFetch() {
176  loading.value = true
177  const { data } = await axios.get('/api/xxx')
178  loading.value = false  // 出错时不执行
179}

184## 五、在组件中使用 Store
185 
186### 正确解构方式
187 
188```typescript
189// ✅ 用 storeToRefs 解构响应式数据，方法直接解构
190const userStore = useUserStore()
191const { userInfo, isLoggedIn } = storeToRefs(userStore)  // 保持响应式
192const { fetchUserInfo, logout } = userStore               // 方法直接取
193 
194// ❌ 直接解构会丢失响应式
195const { userInfo } = useUserStore()  // userInfo 变成普通值，不响应更新
196```
197 
198### 只在需要的组件中调用 store
199 
200```typescript
201// ✅ 在需要的组件内调用
202const userStore = useUserStore()
203 
204// ❌ 在 composable 外部（模块顶层）调用（Pinia 实例未挂载）
205const userStore = useUserStore()  // 模块顶层调用会报错
206 
207export function useXxx() {
208  const userStore = useUserStore()  // ✅ 在函数内部调用
209}
210```

215

217 
218```typescript

221  const theme = useStorage<'light' | 'dark'>('app-theme', 'light')
222  const collapsed = useStorage('sidebar-collapsed', false)
223 
224  return { theme, collapsed }

230## 七、跨 Store 通信
231 
232Store 可以直接调用其他 store，不需要任何特殊处理：
233 
234```typescript
235// stores/order.ts
236export const useOrderStore = defineStore('order', () => {
237  async function submitOrder(form: OrderForm) {
238    // 直接在 action 内调用其他 store
239    const userStore = useUserStore()
240    if (!userStore.isLoggedIn) {
241      message.warning('请先登录')
242      return
243    }
244    await axios.post('/api/order/create', {
245      ...form,
246      userId: userStore.userInfo?.id,
247    })
248  }
249 
250  return { submitOrder }
251})

256## 反模式检查清单
257 
258- [ ] ❌ 把单页面的列表数据放进了 store → 移到组件本地
259- [ ] ❌ 在 store 中直接操作 DOM 或调用路由跳转 → 在组件中处理，store 只管数据
260- [ ] ❌ Store 中有超过 10 个以上的 state 字段 → 考虑拆分成多个 store
261- [ ] ❌ 直接解构 store 的 state 而不用 storeToRefs → 响应式丢失
262- [ ] ❌ 在模块顶层调用 useXxxStore() → 移到函数内部
263- [ ] ❌ 用 store 传递父子组件数据 → 改用 Props / Emits
