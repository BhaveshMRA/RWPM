373// Focus management
374function openModal(modal) {
375  // Store last focused element
376  lastFocus = document.activeElement;
377 
378  // Move focus to modal
379  modal.querySelector("h2").focus();
380 
381  // Trap focus
382  modal.addEventListener("keydown", trapFocus);
383}
384 
385function closeModal(modal) {
386  // Return focus
387  lastFocus.focus();
388}
389 
390function trapFocus(e) {
391  if (e.key === "Tab") {
392    const focusable = modal.querySelectorAll(
393      'button, [href], input, select, textarea, [tabindex]:not([tabindex
 >="-1"])',
394    );
395    const first = focusable[0];
396    const last = focusable[focusable.length - 1];
397 
398    if (e.shiftKey && document.activeElement === first) {
399      last.focus();
400      e.preventDefault();
401    } else if (!e.shiftKey && document.activeElement === last) {
402      first.focus();
403      e.preventDefault();
404    }
405  }
406 
407  if (e.key === "Escape") {
408    closeModal(modal);
409  }
410}

521## Best Practices
522 
523### Do's
524 
525- **Test with actual screen readers** - Not just simulators
526- **Use semantic HTML first** - ARIA is supplemental
527- **Test in browse and focus modes** - Different experiences
528- **Verify focus management** - Especially for SPAs
529- **Test keyboard only first** - Foundation for SR testing
530 
531### Don'ts
532 
533- **Don't assume one SR is enough** - Test multiple
534- **Don't ignore mobile** - Growing user base
535- **Don't test only happy path** - Test error states
536- **Don't skip dynamic content** - Most common issues
537- **Don't rely on visual testing** - Different experience
